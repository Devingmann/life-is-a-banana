    56→/* ---------- tiny geometry helper: tint + transform + merge ---------- */
    57→const _m4 = new THREE.Matrix4();
    58→const _q = new THREE.Quaternion();
    59→const _e = new THREE.Euler();
    60→const _v = new THREE.Vector3();
    61→const _s = new THREE.Vector3();
    62→const _yAxis = new CANNON.Vec3(0, 1, 0);
    63→
    64→function part(geoSrc, hex, pos = [0, 0, 0], rot = [0, 0, 0], scl = [1, 1, 1]) {
    65→  const g = geoSrc.clone();
    66→  _e.set(rot[0], rot[1], rot[2]);
    67→  _q.setFromEuler(_e);
    68→  _v.set(pos[0], pos[1], pos[2]);
    69→  _s.set(scl[0], scl[1], scl[2]);
    70→  _m4.compose(_v, _q, _s);
    71→  g.applyMatrix4(_m4);
    72→  const n = g.attributes.position.count;
    73→  const c = new THREE.Color(hex);
    74→  const arr = new Float32Array(n * 3);
    75→  for (let i = 0; i < n; i++) { arr[i * 3] = c.r; arr[i * 3 + 1] = c.g; arr[i * 3 + 2] = c.b; }
    76→  g.setAttribute('color', new THREE.BufferAttribute(arr, 3));
    77→  return g;
    78→}

   207→  _makeRoadTexture() {
   208→    const c = document.createElement('canvas');
   209→    c.width = 256; c.height = 512;
   210→    const ctx = c.getContext('2d');
   211→    ctx.fillStyle = '#171029';
   212→    ctx.fillRect(0, 0, 256, 512);
   213→    for (let i = 0; i < 1100; i++) {
   214→      ctx.fillStyle = Math.random() < 0.5 ? 'rgba(255,255,255,0.035)' : 'rgba(0,0,0,0.3)';
   215→      ctx.fillRect(Math.random() * 256, Math.random() * 512, 2, 2);
   216→    }
   217→    ctx.strokeStyle = 'rgba(255,46,146,0.22)';
   218→    ctx.lineWidth = 2;
   219→    for (let x = 0; x <= 256; x += 64) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, 512); ctx.stroke(); }
   220→    for (let y = 0; y <= 512; y += 64) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(256, y); ctx.stroke(); }
   221→    ctx.strokeStyle = 'rgba(255,110,199,0.9)';
   222→    ctx.lineWidth = 5;
   223→    ctx.beginPath(); ctx.moveTo(5, 0); ctx.lineTo(5, 512); ctx.stroke();
   224→    ctx.beginPath(); ctx.moveTo(251, 0); ctx.lineTo(251, 512); ctx.stroke();

   226→    ctx.fillRect(124, 24, 8, 110);

   278→  _makeStarGeometry() {
   279→    const shape = new THREE.Shape();
   280→    const outer = 0.52, inner = 0.22;
   281→    for (let i = 0; i < 10; i++) {
   282→      const r = i % 2 === 0 ? outer : inner;
   283→      const a = (i / 10) * Math.PI * 2 - Math.PI / 2;
   284→      const x = Math.cos(a) * r, y = Math.sin(a) * r;
   285→      if (i === 0) shape.moveTo(x, y); else shape.lineTo(x, y);
   286→    }
   287→    shape.closePath();
   288→    const geo = new THREE.ExtrudeGeometry(shape, {
   289→      depth: 0.2, bevelEnabled: true, bevelThickness: 0.05, bevelSize: 0.05, bevelSegments: 1,
   290→    });
   291→    geo.center();
   292→    return geo;
   293→  }

   295→  _buildPalm() {
   296→    const trunk = new THREE.CylinderGeometry(0.16, 0.3, 5.4, 7);
   297→    const coco = new THREE.SphereGeometry(0.22, 8, 6);
   298→    const ring = new THREE.CylinderGeometry(0.33, 0.33, 0.09, 8);
   299→    const frond = new THREE.PlaneGeometry(3.1, 0.8, 4, 1);
   300→    // bend the frond into a droopy leaf
   301→    const fp = frond.attributes.position;
   302→    for (let i = 0; i < fp.count; i++) {
   303→      const x = fp.getX(i);
   304→      fp.setY(i, fp.getY(i) * (1 - Math.pow(Math.abs(x) / 1.55, 2) * 0.55));
   305→      fp.setZ(i, -Math.pow(Math.abs(x) / 1.55, 2) * 0.9);
   306→    }
   307→    frond.computeVertexNormals();
   308→
   309→    const silParts = [part(trunk, COL.trunk, [0, 2.7, 0])];
   310→    const ringParts = [];
   311→    [[1.2], [2.4], [3.6]].forEach(([h]) => ringParts.push(part(ring, 0, [0, h, 0])));
   312→    const cocoParts = [
   313→      part(coco, 0x3a2153, [0.3, 5.25, 0.12]),
   314→      part(coco, 0x3a2153, [-0.25, 5.3, -0.18]),
   315→    ];
   316→    const frondParts = [];
   317→    for (let i = 0; i < 7; i++) {
   318→      const a = (i / 7) * Math.PI * 2;

   328→      color: 0x0e2b1f, emissive: COL.teal, emissiveIntensity: 0.65, roughness: 0.6, side: THREE.DoubleSide,
   329→    });

   579→  reset() {
   580→    this.chunkIndex = 0;
   581→    this.setZone(0, true);
   582→    // clear toast projectiles
   583→    for (let i = this.toasts.length - 1; i >= 0; i--) this._recycleToast(i);
   584→    for (let i = 0; i < this.chunks.length; i++) {
   585→      const chunk = this.chunks[i];
   586→      chunk.zEnd = START_Z - i * CHUNK_LEN;
   587→      this._generate(chunk, 0, i);
   588→    }
   589→    this.nextZ = START_Z - this.chunks.length * CHUNK_LEN;
   590→  }
   591→
   592→  _recycle(chunk, difficulty) {
   593→    chunk.zEnd = this.nextZ;
   594→    this.nextZ -= CHUNK_LEN;
   595→    this._generate(chunk, difficulty, this.chunkIndex++);
   596→  }

   598→  _generate(chunk, D, index) {
   599→    // ---- clear previous content ----
   600→    for (const b of chunk.allBodies) this.physics.removeBody(b);
   601→    chunk.allBodies.length = 0;
   602→    chunk.obstacles.length = 0;
   603→    chunk.stars.length = 0;
   604→    chunk.goldens.length = 0;
   605→    chunk.anims.length = 0;
   606→    for (const d of chunk.disposables) d.dispose();
   607→    chunk.disposables.length = 0;
   608→    chunk.group.clear();
   609→    chunk.gapSpan = null;
   610→
   611→    const zEnd = chunk.zEnd;
   612→    chunk.group.position.z = zEnd; // children use local z in [-CHUNK_LEN, 0]
   613→
   614→    const rng = Math.random;
   615→    const safeRunway = index < 2;
   616→
   617→    // ---- void gap layout (two gaps possible at high difficulty) ----
   618→    let gapLocal = null;
   619→    let gapLocal2 = null;
   620→    if (!safeRunway && index > 3 && rng() < Math.min(0.08 + D * 0.26, 0.38)) {
   621→      const gapLen = Math.min(12.5, 7 + D * 5.5);
   622→      const gz = -(12 + rng() * (CHUNK_LEN - 26 - gapLen));
   623→      gapLocal = { near: gz, far: gz - gapLen };
   624→      if (D > 0.55 && rng() < 0.4) {
   625→        const gz2 = -(Math.max(6, gapLocal.far - 8) - (6 + rng() * 8));
   626→        if (gz2 - 6 > -CHUNK_LEN) gapLocal2 = { near: gz2, far: Math.max(-CHUNK_LEN + 2, gz2 - gapLen * 0.8) };
   627→      }
   628→    }
   629→
   630→    // ---- solid road segments ----
   631→    const segments = [];
   632→    if (gapLocal2 && gapLocal.far > gapLocal2.near) {
   633→      segments.push({ z0: 0, z1: gapLocal.near });
   634→      segments.push({ z0: gapLocal.far, z1: gapLocal2.near });
   635→      segments.push({ z0: gapLocal2.far, z1: -CHUNK_LEN });
   636→    } else if (gapLocal) {
   637→      segments.push({ z0: 0, z1: gapLocal.near });
   638→      segments.push({ z0: gapLocal.far, z1: -CHUNK_LEN });
   639→    } else {
   640→      segments.push({ z0: 0, z1: -CHUNK_LEN });
   641→    }
   642→    for (const seg of segments) {
   643→      if (seg.z0 - seg.z1 < 1) continue; // skip degenerate segments
   644→      const len = seg.z0 - seg.z1;
   645→      const mid = (seg.z0 + seg.z1) / 2;
   646→      const worldMid = zEnd + mid;
   647→      // road plane
   648→      const geo = new THREE.PlaneGeometry(ROAD_W, len);
   649→      const uv = geo.attributes.uv;
   650→      for (let i = 0; i < uv.count; i++) uv.setY(i, uv.getY(i) * (len / 10));
   651→      const road = new THREE.Mesh(geo, this.matRoad);
   652→      road.rotation.x = -Math.PI / 2;
   653→      road.position.set(0, 0, mid);
   654→      road.receiveShadow = true;
   655→      chunk.group.add(road);
   656→      chunk.disposables.push(geo);
   657→      // physics slab (top at y=0)
   658→      chunk.allBodies.push(this.physics.addStaticBox(
   659→        { x: ROAD_W / 2, y: 0.5, z: len / 2 }, { x: 0, y: -0.5, z: worldMid }, { kind: 'ground' }
   660→      ));
   661→      // neon edge strips
   662→      for (const sx of [-ROAD_W / 2 + 0.15, ROAD_W / 2 - 0.15]) {
   663→        const strip = new THREE.Mesh(this.geoEdge, this.matEdge);
   664→        strip.scale.set(0.3, 0.17, len);
   665→        strip.position.set(sx, 0.08, mid);
   666→        chunk.group.add(strip);
   667→      }
   668→    }
   669→
   670→    // gap warning strips + span bookkeeping
   671→    if (gapLocal) {
   672→      for (const gz of [gapLocal.near + 0.4, gapLocal.far - 0.4]) {
   673→        const w = new THREE.Mesh(this.geoWarn, this.matWarn);
   674→        w.position.set(0, 0.045, gz);
   675→        chunk.group.add(w);
   676→      }
   677→      chunk.gapSpan = { zNear: zEnd + gapLocal.near, zFar: zEnd + gapLocal.far };
   678→    }
   679→
   680→    if (safeRunway) {
   681→      this._addScenery(chunk, rng);
   682→      this._addStarLine(chunk, rng, D, null);
   683→      return;
   684→    }
   685→
   686→    // ---- slip zones (banana peel traps) ----
   687→    const slipZones = [];
   688→    if (rng() < 0.22 + D * 0.18) {
   689→      const patchZ = -(8 + rng() * (CHUNK_LEN - 18));
   690→      if (!this._inGap(gapLocal, patchZ, 6)) {
   691→        const px = (rng() - 0.5) * 6.4;
   692→        const half = { x: 2.7, y: 0.09, z: 4.2 };
   693→        const body = this.physics.addStaticBox(half, { x: px, y: 0.1, z: zEnd + patchZ }, {
   694→          kind: 'slipzone', material: this.physics.groundMaterial, userData: { chunk, id: rng() },
   695→        });
   696→        chunk.allBodies.push(body);
   697→        slipZones.push(patchZ);
   698→        // scattered peel bits
   699→        const nPeels = 7 + ((rng() * 3) | 0);
   700→        for (let i = 0; i < nPeels; i++) {
   701→          const peel = new THREE.Mesh(this.geoPeel, this.matPeel);
   702→          peel.position.set(px + (rng() - 0.5) * 4.6, 0.06, patchZ + (rng() - 0.5) * 7);
   703→          peel.rotation.set(0, rng() * Math.PI * 2, (rng() - 0.5) * 0.25);
   704→          peel.scale.setScalar(0.8 + rng() * 0.5);
   705→          chunk.group.add(peel);
   706→        }
   707→      }
   708→    }
   709→
   710→    // ---- speed pads (turbo chevrons — positive interaction) ----
   711→    if (D > 0.12 && rng() < 0.5) {
   712→      const padZ = -(6 + rng() * (CHUNK_LEN - 14));
   713→      if (!this._inGap(gapLocal, padZ, 5) && !slipZones.some((s) => Math.abs(s - padZ) < 7)) {
   714→        const padX = (rng() - 0.5) * 7;
   715→        const pad = new THREE.Mesh(this.geoPad, this.matPad);
   716→        pad.position.set(padX, 0.05, padZ);
   717→        chunk.group.add(pad);
   718→        const body = this.physics.addStaticBox(
   719→          { x: 1.35, y: 0.05, z: 2.3 }, { x: padX, y: 0.05, z: zEnd + padZ },
   720→          { kind: 'speedpad', material: this.physics.groundMaterial, userData: { chunk } }
   721→        );
   722→        chunk.allBodies.push(body);
   723→      }
   724→    }
   725→
   726→    // ---- obstacles (denser as difficulty climbs) ----
   727→    const count = Math.min(6, 2 + Math.round(D * 4.2) + (D > 0.55 && rng() < 0.4 ? 1 : 0));
   728→    const minSpace = Math.max(6.7, 8.5 - D * 1.8);
   729→    const placed = [];
   730→    for (let i = 0; i < count; i++) {
   731→      let lz = null;
   732→      for (let attempt = 0; attempt < 26; attempt++) {
   733→        const z = -(5 + rng() * (CHUNK_LEN - 10));
   734→        if (this._inGap(gapLocal, z, 7)) continue;
   735→        if (slipZones.some((s) => Math.abs(s - z) < 6)) continue;
   736→        if (placed.some((s) => Math.abs(s - z) < minSpace)) continue;
   737→        lz = z;
   738→        break;
   739→      }
   740→      if (lz === null) continue;
   741→      placed.push(lz);
   742→      const type = this._pickObstacleType(D, rng, placed.length === 1);
   743→      this._addObstacle(chunk, type, lz, rng, D);
   744→    }
   745→
   746→    // ---- potassium star lines ----
   747→    if (rng() < 0.75) this._addStarLine(chunk, rng, D, gapLocal);
   748→
   749→    // ---- rare golden banana jackpot ----
   750→    if (rng() < 0.16 + D * 0.08) this._addGolden(chunk, rng, gapLocal);
   751→
   752→    // ---- scenery ----
   753→    this._addScenery(chunk, rng);
   754→  }

   756→  _inGap(gapLocal, z, margin) {
   757→    return !!gapLocal && z < gapLocal.near + margin && z > gapLocal.far - margin;
   758→  }

   760→  _pickObstacleType(D, rng, first) {
   761→    const entries = [
   762→      ['monkey', 10],
   763→      ['toaster', D > 0.08 ? 7 : 0],
   764→      ['blender', D > 0.18 ? 4 + D * 5 : 0],
   765→      ['gorilla', D > 0.3 && first ? 3.5 : 0],
   766→      ['juicer', D > 0.3 ? 3 + D * 6 : 0],
   767→    ];
   768→    let total = 0;
   769→    for (const [, w] of entries) total += w;
   770→    let r = rng() * total;
   771→    for (const [t, w] of entries) { r -= w; if (r <= 0) return t; }
   772→    return 'monkey';
   773→  }

   775→  _addObstacle(chunk, type, lz, rng, D) {
   776→    const zEnd = chunk.zEnd;
   777→    let mesh, body, damage = 1, knock = 7, extraBody = null, nmDist = 2.05;
   778→
   779→    if (type === 'monkey') {
   780→      mesh = new THREE.Mesh(this.geoMonkey, this.matFlat);
   781→      mesh.castShadow = true;
   782→      const x = (rng() - 0.5) * 8.8;
   783→      mesh.position.set(x, 0, lz);
   784→      const phase = rng() * 9;
   785→      // PATROL MONKEYS: cross the road at higher difficulty — read the lane!
   786→      const moving = D > 0.22 && rng() < 0.45;
   787→      if (moving) {
   788→        const amp = Math.min(4.2, 2.4 + D * 1.8);
   789→        const spd = 0.65 + D * 0.7 + rng() * 0.25;
   790→        const baseX = THREE.MathUtils.clamp(x, -ROAD_HALF * 0.55 + 1, ROAD_HALF * 0.55 - 1) * 0.4;
   791→        const mphase = rng() * Math.PI * 2;
   792→        body = this.physics.addStaticBox({ x: 0.62, y: 0.75, z: 0.6 }, { x, y: 0.75, z: zEnd + lz }, { userData: { type, moving: true } });
   793→        chunk.anims.push((t) => {
   794→          const mx = baseX + Math.sin(t * spd + mphase) * amp;
   795→          mesh.position.x = THREE.MathUtils.clamp(mx, -5.2, 5.2);
   796→          mesh.position.y = Math.abs(Math.sin(t * 2.2 + phase)) * 0.22;
   797→          mesh.rotation.z = Math.sin(t * 1.6 + phase) * 0.07;
   798→          body.position.x = mesh.position.x;
   799→        });
   800→      } else {
   801→        chunk.anims.push((t) => {
   802→          mesh.position.y = Math.abs(Math.sin(t * 2.2 + phase)) * 0.22;
   803→          mesh.rotation.z = Math.sin(t * 1.6 + phase) * 0.07;
   804→        });
   805→        body = this.physics.addStaticBox({ x: 0.62, y: 0.75, z: 0.6 }, { x, y: 0.75, z: zEnd + lz }, { userData: { type } });
   806→      }
   807→    } else if (type === 'gorilla') {
   808→      mesh = new THREE.Group();
   809→      const g = new THREE.Mesh(this.geoGorilla, this.matFlat);
   810→      g.castShadow = true;
   811→      mesh.add(g);
   812→      const arms = [];
   813→      for (const sx of [0.95, -0.95]) {
   814→        const arm = new THREE.Group();
   815→        const am = new THREE.Mesh(this.geoGorillaArm, this.matGorillaArm);
   816→        am.position.y = -0.65;
   817→        am.castShadow = true;
   818→        const fist = new THREE.Mesh(this.geoGorillaFist, this.matGorillaArm);
   819→        fist.position.y = -1.35;
   820→        arm.add(am, fist);

   930→  _addStarLine(chunk, rng, D, gapLocal) {
   931→    const n = 6 + ((rng() * 3) | 0);
   932→    const pattern = rng();
   933→    const startX = (rng() - 0.5) * 8;
   934→    const startZ = -(6 + rng() * (CHUNK_LEN - 16));
   935→    const spacing = 2.3;
   936→    for (let i = 0; i < n; i++) {
   937→      let x = startX, y = 1.35, z = startZ - i * spacing;
   938→      if (pattern < 0.30) {
   939→        // classic arc (floats over gaps)
   940→        y = 1.35 + Math.sin((i / (n - 1)) * Math.PI) * 2.2;
   941→        x = startX + Math.sin((i / (n - 1)) * Math.PI) * 1.6;
   942→      } else if (pattern < 0.55) {
   943→        // sine weave
   944→        x = startX + Math.sin(i * 0.85) * 2.4;
   945→      } else if (pattern < 0.72) {
   946→        // tight zigzag (harder, richer)
   947→        y = 1.35 + Math.sin((i / (n - 1)) * Math.PI) * 1.2;
   948→        x = startX + ((i % 2 === 0) ? -2.7 : 2.7);
   949→      } else if (pattern < 0.86) {
   950→        // double arc over a longer span
   951→        y = 1.35 + Math.abs(Math.sin((i / (n - 1)) * Math.PI * 2)) * 2.4;
   952→        x = startX + Math.sin((i / (n - 1)) * Math.PI * 2) * 2.0;
   953→      }
   954→      // ground-hugging patterns (weave / zigzag) skip gap sections;
   955→      // arcs and double arcs are high enough to float over gaps
   956→      const floatsOverGap = pattern < 0.30 || (pattern >= 0.72 && pattern < 0.86);
   957→      if (!floatsOverGap && this._inGap(gapLocal, z, 1)) continue;
   958→      const mesh = new THREE.Mesh(this.geoStar, this.matStar);
   959→      const phase = rng() * 9;
   960→      const baseY = y;
   961→      mesh.position.set(x, y, z);
   962→      mesh.rotation.z = rng() * Math.PI;
   963→      chunk.group.add(mesh);
   964→      const star = { mesh, wx: x, wy: y, wz: chunk.zEnd + z, baseY, phase, collected: false };
   965→      chunk.stars.push(star);
   966→      chunk.anims.push((t, dt) => {
   967→        if (star.collected) return;
   968→        mesh.rotation.y += 2.6 * dt;
   969→        mesh.position.y = star.baseY + Math.sin(t * 2.4 + phase) * 0.26;
   970→        star.wy = mesh.position.y;
   971→      });
   972→    }

  1078→  _updateToasts(dt, ctx, px, py, pz) {
  1079→    // launch from toasters ahead of the player at high difficulty
  1080→    if (ctx.difficulty > 0.42 && ctx.canCollect) {
  1081→      for (const chunk of this.chunks) {
  1082→        for (const ob of chunk.obstacles) {
  1083→          if (ob.dead || !ob.body.userData || ob.body.userData.type !== 'toaster') continue;
  1084→          ob.launchT -= dt;
  1085→          const wz = ob.mesh.position.z + chunk.group.position.z;
  1086→          const ahead = pz - wz;
  1087→          if (ob.launchT <= 0 && ahead > 6 && ahead < 30 && this.toasts.length < 10) {
  1088→            ob.launchT = 2.6 + Math.random() * 2.2;
  1089→            this._launchToast(ob.mesh.position.x, wz, px, pz);
  1090→          }
  1091→        }
  1092→      }
  1093→    }
  1094→    // integrate + hit test
  1095→    for (let i = this.toasts.length - 1; i >= 0; i--) {
  1096→      const t = this.toasts[i];
  1097→      t.life -= dt;
  1098→      t.vy -= 24 * dt;
  1099→      t.x += t.vx * dt; t.y += t.vy * dt; t.z += t.vz * dt;
  1100→      t.mesh.position.set(t.x, t.y, t.z);
  1101→      t.mesh.rotation.x += t.spin * dt;
  1102→      t.mesh.rotation.z += t.spin * 0.7 * dt;
  1103→      const dx = t.x - px, dy = t.y - py, dz = t.z - pz;
  1104→      if (ctx.canCollect && dx * dx + dy * dy + dz * dz < 2.2) {
  1105→        if (ctx.onToastHit) ctx.onToastHit();
  1106→        this._recycleToast(i);
  1107→        continue;
  1108→      }
  1109→      if (t.life <= 0 || t.y < -0.4) this._recycleToast(i);
  1110→    }
  1111→  }

  1149→  update(dt, ctx) {
  1150→    this.time += dt;
  1151→    const t = ctx.time;
  1152→    const pz = ctx.playerPos.z;
  1153→    const px = ctx.playerPos.x;
  1154→    const py = ctx.playerPos.y;
  1155→
  1156→    // ---- zone palette lerp ----
  1157→    this._lerpZone(dt);
  1158→
  1159→    // ---- recycle chunks that fell behind ----
  1160→    for (const chunk of this.chunks) {
  1161→      if (chunk.zEnd - CHUNK_LEN > pz + 26) this._recycle(chunk, ctx.difficulty);
  1162→    }
  1163→
  1164→    // ---- animate & collect ----
  1165→    for (const chunk of this.chunks) {
  1166→      for (const anim of chunk.anims) anim(t, dt);
  1167→      if (!ctx.canCollect) continue;
  1168→      for (const star of chunk.stars) {
  1169→        if (star.collected) continue;
  1170→        const dx = star.wx - px, dy = star.wy - py, dz = star.wz - pz;
  1171→        const d2 = dx * dx + dy * dy + dz * dz;
  1172→        if (ctx.superMode && d2 < 90) {
  1173→          const d = Math.sqrt(d2) || 1;
  1174→          const k = Math.min(1, (dt * 20) / d);
  1175→          star.wx += (px - star.wx) * k;
  1176→          star.wy += (py - star.wy) * k;
  1177→          star.wz += (pz - star.wz) * k;
  1178→          star.mesh.position.set(star.wx, star.wy, star.wz - chunk.zEnd);
  1179→          if (d < 2.4) this._collect(star, ctx);
  1180→        } else if (d2 < 2.9) {
  1181→          this._collect(star, ctx);
  1182→        }
  1183→      }
  1184→      // golden bananas (bigger radius + super magnet)
  1185→      for (const g of chunk.goldens) {
  1186→        if (g.collected) continue;
  1187→        const dx = g.wx - px, dy = g.wy - py, dz = g.wz - pz;
  1188→        const d2 = dx * dx + dy * dy + dz * dz;
  1189→        if (ctx.superMode && d2 < 42) {
  1190→          const d = Math.sqrt(d2) || 1;
  1191→          const k = Math.min(1, (dt * 16) / d);
  1192→          g.wx += (px - g.wx) * k;
  1193→          g.wy += (py - g.wy) * k;
  1194→          g.wz += (pz - g.wz) * k;
  1195→          g.mesh.position.set(g.wx, g.wy, g.wz - chunk.zEnd);
  1196→        }
  1197→        if (d2 < 3.4) {
  1198→          g.collected = true;
  1199→          g.mesh.visible = false;
  1200→          if (ctx.onGolden) ctx.onGolden(g);
  1201→        }
  1202→      }
  1203→    }
  1204→
  1205→    // ---- near-miss GRAZE detection (score for style) ----
  1206→    if (ctx.nearMissOK && ctx.onNearMiss) {
  1207→      for (const chunk of this.chunks) {
  1208→        for (const ob of chunk.obstacles) {
  1209→          if (ob.dead || ob.nearMissed) continue;
  1210→          const wz = ob.mesh.position.z + chunk.group.position.z;
  1211→          const rel = pz - wz;
  1212→          if (rel > 3.2) { ob.nearMissed = true; continue; } // passed cleanly
  1213→          if (rel > 0 && rel < 2.4) {
  1214→            const dx = Math.abs(px - ob.mesh.position.x);
  1215→            if (dx < (ob.body.userData.nmDist || 2.05)) {
  1216→              ob.nearMissed = true;
  1217→              ctx.onNearMiss(px, py + 0.6, pz);
  1218→            }
  1219→          }
  1220→        }
  1221→      }
  1222→    }
  1223→
  1224→    // ---- toast barrage ----
  1225→    this._updateToasts(dt, ctx, px, py, pz);
  1226→
  1227→    // ---- world-fixed grid floor follows the runner ----
  1228→    this.grid.position.z = pz;
  1229→    this.gridTex.offset.y = -pz / 10;
  1230→
  1231→    // ---- sky follows camera ----
  1232→    const cp = ctx.camera.position;
  1233→    this.sun.position.set(cp.x * 0.25, 30 + cp.y * 0.1, cp.z - 300);
  1234→    this.starfield.position.copy(cp);
  1235→  }
