    56→/* ---------- tiny geometry helper: tint + transform + merge ---------- */
    57→const _m4 = new THREE.Matrix4();
    58→const _q = new THREE.Quaternion();
    59→const _e = new THREE.Euler();
    60→const _v = new THREE.Vector3();
    61→const _s = new THREE.Vector3();
    62→const _yAxis = new CANNON.Vec3(0, 1, 0);
    63→
    64→function part(geoSrc, hex, pos = [0, 0, 0], rot = [0, 0, 0], scl = [1, 1, 1]) {
    65→  const g = geoSrc.clone();
    66→  _e.set(rot[0], rot[1], rot[2]);
    67→  _q.setFromEuler(_e);
    68→  _v.set(pos[0], pos[1], pos[2]);
    69→  _s.set(scl[0], scl[1], scl[2]);
    70→  _m4.compose(_v, _q, _s);
    71→  g.applyMatrix4(_m4);
    72→  const n = g.attributes.position.count;
    73→  const c = new THREE.Color(hex);
    74→  const arr = new Float32Array(n * 3);
    75→  for (let i = 0; i < n; i++) { arr[i * 3] = c.r; arr[i * 3 + 1] = c.g; arr[i * 3 + 2] = c.b; }
    76→  g.setAttribute('color', new THREE.BufferAttribute(arr, 3));
    77→  return g;
    78→}

   206→  _makeRoadTexture() {
   207→    // RETRO KITSCH asphalt: dark purple base, hot-pink grid, chunky yellow
   208→    // dashes and black & white checkerboard shoulders — a proper outrun
   209→    // arcade carpet, readable at any speed.
   210→    const c = document.createElement('canvas');
   211→    c.width = 256; c.height = 512;
   212→    const ctx = c.getContext('2d');
   213→    ctx.fillStyle = '#171029';
   214→    ctx.fillRect(0, 0, 256, 512);
   215→    for (let i = 0; i < 1100; i++) {
   216→      ctx.fillStyle = Math.random() < 0.5 ? 'rgba(255,255,255,0.035)' : 'rgba(0,0,0,0.3)';
   217→      ctx.fillRect(Math.random() * 256, Math.random() * 512, 2, 2);
   218→    }
   219→    // pink lane grid
   220→    ctx.strokeStyle = 'rgba(255,46,146,0.22)';
   221→    ctx.lineWidth = 2;
   222→    for (let x = 0; x <= 256; x += 64) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, 512); ctx.stroke(); }
   223→    for (let y = 0; y <= 512; y += 64) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(256, y); ctx.stroke(); }
   224→    // checkerboard shoulders (the classic kitsch racing-rim carpet)
   225→    const cell = 16;
   226→    for (const x0 of [0, 256 - 32]) {
   227→      for (let y = 0; y < 512; y += cell) {
   228→        for (let x = 0; x < 32; x += cell) {
   229→          const odd = ((x / cell) + (y / cell)) % 2 === 0;
   230→          ctx.fillStyle = odd ? '#e9edff' : '#221643';
   231→          ctx.fillRect(x0 + x, y, cell, cell);
   232→        }
   233→      }
   234→    }
   235→    // pink boundary lines between checker and asphalt
   236→    ctx.strokeStyle = 'rgba(255,110,199,0.95)';
   237→    ctx.lineWidth = 5;
   238→    ctx.beginPath(); ctx.moveTo(35, 0); ctx.lineTo(35, 512); ctx.stroke();
   239→    ctx.beginPath(); ctx.moveTo(221, 0); ctx.lineTo(221, 512); ctx.stroke();
   240→    // chunky yellow centre dashes
   241→    ctx.fillStyle = 'rgba(255,225,53,0.85)';
   242→    ctx.fillRect(124, 24, 9, 110);
   243→    ctx.fillRect(124, 276, 9, 110);
   244→    const tex = new THREE.CanvasTexture(c);
   245→    tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
   246→    tex.colorSpace = THREE.SRGBColorSpace;
   247→    tex.anisotropy = 4;
   248→    return tex;
   249→  }

   294→  _makeStarGeometry() {
   295→    // chunkier arcade star: fat points, thick body, deep bevel
   296→    const shape = new THREE.Shape();
   297→    const outer = 0.56, inner = 0.235;
   298→    for (let i = 0; i < 10; i++) {
   299→      const r = i % 2 === 0 ? outer : inner;
   300→      const a = (i / 10) * Math.PI * 2 - Math.PI / 2;
   301→      const x = Math.cos(a) * r, y = Math.sin(a) * r;
   302→      if (i === 0) shape.moveTo(x, y); else shape.lineTo(x, y);
   303→    }
   304→    shape.closePath();
   305→    const geo = new THREE.ExtrudeGeometry(shape, {
   306→      depth: 0.24, bevelEnabled: true, bevelThickness: 0.07, bevelSize: 0.07, bevelSegments: 2,
   307→    });
   308→    geo.center();
   309→    return geo;
   310→  }

   312→  _buildPalm() {
   313→    const trunk = new THREE.CylinderGeometry(0.16, 0.3, 5.4, 7);
   314→    const coco = new THREE.SphereGeometry(0.22, 8, 6);
   315→    const ring = new THREE.CylinderGeometry(0.33, 0.33, 0.09, 8);
   316→    const frond = new THREE.PlaneGeometry(3.1, 0.8, 4, 1);
   317→    // bend the frond into a droopy leaf
   318→    const fp = frond.attributes.position;
   319→    for (let i = 0; i < fp.count; i++) {
   320→      const x = fp.getX(i);
   321→      fp.setY(i, fp.getY(i) * (1 - Math.pow(Math.abs(x) / 1.55, 2) * 0.55));
   322→      fp.setZ(i, -Math.pow(Math.abs(x) / 1.55, 2) * 0.9);
   323→    }
   324→    frond.computeVertexNormals();
   325→
   326→    const silParts = [part(trunk, COL.trunk, [0, 2.7, 0])];
   327→    const ringParts = [];
   328→    [[1.2], [2.4], [3.6]].forEach(([h]) => ringParts.push(part(ring, 0, [0, h, 0])));
   329→    const cocoParts = [
   330→      part(coco, 0x3a2153, [0.3, 5.25, 0.12]),
   331→      part(coco, 0x3a2153, [-0.25, 5.3, -0.18]),
   332→    ];
   333→    const frondParts = [];
   334→    for (let i = 0; i < 7; i++) {
   335→      const a = (i / 7) * Math.PI * 2;
   336→      const dir = [Math.cos(a) * 1.35, 5.4, Math.sin(a) * 1.35];
   337→      frondParts.push(part(frond, 0, dir, [0, -a, -0.42], [1, 1, 1]));
   338→    }
   339→    this.geoPalmSil = mergeGeometries([...silParts, ...cocoParts], false);
   340→    this.geoPalmRings = mergeGeometries(ringParts, false);
   341→    this.geoPalmFronds = mergeGeometries(frondParts, false);
   342→    this.matPalmSil = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.9 });
   343→    this.matPalmRings = new THREE.MeshStandardMaterial({ color: COL.pink, emissive: COL.magenta, emissiveIntensity: 0.9 });
   344→    this.matPalmFronds = new THREE.MeshStandardMaterial({
   345→      color: 0x0e2b1f, emissive: COL.teal, emissiveIntensity: 0.55, roughness: 0.6, side: THREE.DoubleSide,
   346→    });
   347→  }

   595→
   596→  reset() {
   597→    this.chunkIndex = 0;
   598→    this.setZone(0, true);
   599→    // clear toast projectiles
   600→    for (let i = this.toasts.length - 1; i >= 0; i--) this._recycleToast(i);
   601→    for (let i = 0; i < this.chunks.length; i++) {
   602→      const chunk = this.chunks[i];
   603→      chunk.zEnd = START_Z - i * CHUNK_LEN;
   604→      this._generate(chunk, 0, i);
   605→    }
   606→    this.nextZ = START_Z - this.chunks.length * CHUNK_LEN;
   607→  }
   608→
   609→  _recycle(chunk, difficulty) {
   610→    chunk.zEnd = this.nextZ;
   611→    this.nextZ -= CHUNK_LEN;
   612→    this._generate(chunk, difficulty, this.chunkIndex++);
   613→  }
   614→
   615→  _generate(chunk, D, index) {
   616→    // ---- clear previous content ----
   617→    for (const b of chunk.allBodies) this.physics.removeBody(b);
   618→    chunk.allBodies.length = 0;
   619→    chunk.obstacles.length = 0;
   620→    chunk.stars.length = 0;
   621→    chunk.goldens.length = 0;
   622→    chunk.anims.length = 0;
   623→    for (const d of chunk.disposables) d.dispose();
   624→    chunk.disposables.length = 0;
   625→    chunk.group.clear();
   626→    chunk.gapSpan = null;
   627→
   628→    const zEnd = chunk.zEnd;
   629→    chunk.group.position.z = zEnd; // children use local z in [-CHUNK_LEN, 0]
   630→
   631→    const rng = Math.random;
   632→    const safeRunway = index < 2;
   633→
   634→    // ---- void gap layout (two gaps possible at high difficulty) ----
   635→    let gapLocal = null;
   636→    let gapLocal2 = null;
   637→    if (!safeRunway && index > 3 && rng() < Math.min(0.08 + D * 0.26, 0.38)) {
   638→      const gapLen = Math.min(12.5, 7 + D * 5.5);
   639→      const gz = -(12 + rng() * (CHUNK_LEN - 26 - gapLen));
   640→      gapLocal = { near: gz, far: gz - gapLen };
   641→      if (D > 0.55 && rng() < 0.4) {
   642→        const gz2 = -(Math.max(6, gapLocal.far - 8) - (6 + rng() * 8));
   643→        if (gz2 - 6 > -CHUNK_LEN) gapLocal2 = { near: gz2, far: Math.max(-CHUNK_LEN + 2, gz2 - gapLen * 0.8) };
   644→      }
   645→    }
   646→
   647→    // ---- solid road segments ----
   648→    const segments = [];
   649→    if (gapLocal2 && gapLocal.far > gapLocal2.near) {
   650→      segments.push({ z0: 0, z1: gapLocal.near });
   651→      segments.push({ z0: gapLocal.far, z1: gapLocal2.near });
   652→      segments.push({ z0: gapLocal2.far, z1: -CHUNK_LEN });
   653→    } else if (gapLocal) {
   654→      segments.push({ z0: 0, z1: gapLocal.near });
   655→      segments.push({ z0: gapLocal.far, z1: -CHUNK_LEN });
   656→    } else {
   657→      segments.push({ z0: 0, z1: -CHUNK_LEN });
   658→    }
   659→    for (const seg of segments) {
   660→      if (seg.z0 - seg.z1 < 1) continue; // skip degenerate segments
   661→      const len = seg.z0 - seg.z1;
   662→      const mid = (seg.z0 + seg.z1) / 2;
   663→      const worldMid = zEnd + mid;
   664→      // road plane
   665→      const geo = new THREE.PlaneGeometry(ROAD_W, len);
   666→      const uv = geo.attributes.uv;
   667→      for (let i = 0; i < uv.count; i++) uv.setY(i, uv.getY(i) * (len / 10));
   668→      const road = new THREE.Mesh(geo, this.matRoad);
   669→      road.rotation.x = -Math.PI / 2;
   670→      road.position.set(0, 0, mid);
   671→      road.receiveShadow = true;
   672→      chunk.group.add(road);
   673→      chunk.disposables.push(geo);
   674→      // physics slab (top at y=0)
   675→      chunk.allBodies.push(this.physics.addStaticBox(
   676→        { x: ROAD_W / 2, y: 0.5, z: len / 2 }, { x: 0, y: -0.5, z: worldMid }, { kind: 'ground' }
   677→      ));
   678→      // neon edge strips
   679→      for (const sx of [-ROAD_W / 2 + 0.15, ROAD_W / 2 - 0.15]) {
   680→        const strip = new THREE.Mesh(this.geoEdge, this.matEdge);
   681→        strip.scale.set(0.3, 0.17, len);
   682→        strip.position.set(sx, 0.08, mid);
   683→        chunk.group.add(strip);
   684→      }
   685→    }
   686→
   687→    // gap warning strips + span bookkeeping
   688→    if (gapLocal) {
   689→      for (const gz of [gapLocal.near + 0.4, gapLocal.far - 0.4]) {
   690→        const w = new THREE.Mesh(this.geoWarn, this.matWarn);
   691→        w.position.set(0, 0.045, gz);
   692→        chunk.group.add(w);
   693→      }
   694→      chunk.gapSpan = { zNear: zEnd + gapLocal.near, zFar: zEnd + gapLocal.far };
   695→    }
   696→
   697→    if (safeRunway) {
   698→      this._addScenery(chunk, rng);
   699→      this._addStarLine(chunk, rng, D, null);
   700→      return;
   701→    }
   702→
   703→    // ---- slip zones (banana peel traps) ----
   704→    const slipZones = [];
   705→    if (rng() < 0.22 + D * 0.18) {
   706→      const patchZ = -(8 + rng() * (CHUNK_LEN - 18));
   707→      if (!this._inGap(gapLocal, patchZ, 6)) {
   708→        const px = (rng() - 0.5) * 6.4;
   709→        const half = { x: 2.7, y: 0.09, z: 4.2 };
   710→        const body = this.physics.addStaticBox(half, { x: px, y: 0.1, z: zEnd + patchZ }, {
   711→          kind: 'slipzone', material: this.physics.groundMaterial, userData: { chunk, id: rng() },
   712→        });
   713→        chunk.allBodies.push(body);
   714→        slipZones.push(patchZ);
   715→        // scattered peel bits
   716→        const nPeels = 7 + ((rng() * 3) | 0);
   717→        for (let i = 0; i < nPeels; i++) {
   718→          const peel = new THREE.Mesh(this.geoPeel, this.matPeel);
   719→          peel.position.set(px + (rng() - 0.5) * 4.6, 0.06, patchZ + (rng() - 0.5) * 7);
   720→          peel.rotation.set(0, rng() * Math.PI * 2, (rng() - 0.5) * 0.25);
   721→          peel.scale.setScalar(0.8 + rng() * 0.5);
   722→          chunk.group.add(peel);
   723→        }
   724→      }
   725→    }
   726→
   727→    // ---- speed pads (turbo chevrons — positive interaction) ----
   728→    if (D > 0.12 && rng() < 0.5) {
   729→      const padZ = -(6 + rng() * (CHUNK_LEN - 14));
   730→      if (!this._inGap(gapLocal, padZ, 5) && !slipZones.some((s) => Math.abs(s - padZ) < 7)) {
   731→        const padX = (rng() - 0.5) * 7;
   732→        const pad = new THREE.Mesh(this.geoPad, this.matPad);
   733→        pad.position.set(padX, 0.05, padZ);
   734→        chunk.group.add(pad);
   735→        const body = this.physics.addStaticBox(
   736→          { x: 1.35, y: 0.05, z: 2.3 }, { x: padX, y: 0.05, z: zEnd + padZ },
   737→          { kind: 'speedpad', material: this.physics.groundMaterial, userData: { chunk } }
   738→        );
   739→        chunk.allBodies.push(body);
   740→      }
   741→    }
   742→
   743→    // ---- obstacles (denser as difficulty climbs) ----
   744→    const count = Math.min(6, 2 + Math.round(D * 4.2) + (D > 0.55 && rng() < 0.4 ? 1 : 0));
   745→    const minSpace = Math.max(6.7, 8.5 - D * 1.8);
   746→    const placed = [];
   747→    for (let i = 0; i < count; i++) {
   748→      let lz = null;
   749→      for (let attempt = 0; attempt < 26; attempt++) {
   750→        const z = -(5 + rng() * (CHUNK_LEN - 10));
   751→        if (this._inGap(gapLocal, z, 7)) continue;
   752→        if (slipZones.some((s) => Math.abs(s - z) < 6)) continue;
   753→        if (placed.some((s) => Math.abs(s - z) < minSpace)) continue;
   754→        lz = z;
   755→        break;
   756→      }
   757→      if (lz === null) continue;
   758→      placed.push(lz);
   759→      const type = this._pickObstacleType(D, rng, placed.length === 1);
   760→      this._addObstacle(chunk, type, lz, rng, D);
   761→    }
   762→
   763→    // ---- potassium star lines ----
   764→    if (rng() < 0.75) this._addStarLine(chunk, rng, D, gapLocal);
   765→
   766→    // ---- rare golden banana jackpot ----
   767→    if (rng() < 0.16 + D * 0.08) this._addGolden(chunk, rng, gapLocal);
   768→
   769→    // ---- scenery ----
   770→    this._addScenery(chunk, rng);
   771→  }
   772→
   773→  _inGap(gapLocal, z, margin) {
   774→    return !!gapLocal && z < gapLocal.near + margin && z > gapLocal.far - margin;
   775→  }
   776→
   777→  _pickObstacleType(D, rng, first) {
   778→    const entries = [
   779→      ['monkey', 10],
   780→      ['toaster', D > 0.08 ? 7 : 0],
   781→      ['blender', D > 0.18 ? 4 + D * 5 : 0],
   782→      ['gorilla', D > 0.3 && first ? 3.5 : 0],
   783→      ['juicer', D > 0.3 ? 3 + D * 6 : 0],
   784→    ];
   785→    let total = 0;
   786→    for (const [, w] of entries) total += w;
   787→    let r = rng() * total;
   788→    for (const [t, w] of entries) { r -= w; if (r <= 0) return t; }
   789→    return 'monkey';
   790→  }
   791→
   792→  _addObstacle(chunk, type, lz, rng, D) {
   793→    const zEnd = chunk.zEnd;
   794→    let mesh, body, damage = 1, knock = 7, extraBody = null, nmDist = 2.05, hitC = null;
   795→
   796→    if (type === 'monkey') {
   797→      mesh = new THREE.Mesh(this.geoMonkey, this.matFlat);
   798→      mesh.castShadow = true;
   799→      const x = (rng() - 0.5) * 8.8;
   800→      mesh.position.set(x, 0, lz);
   801→      const phase = rng() * 9;
   802→      // PATROL MONKEYS: cross the road at higher difficulty — read the lane!
   803→      const moving = D > 0.22 && rng() < 0.45;
   804→      if (moving) {
   805→        const amp = Math.min(4.2, 2.4 + D * 1.8);
   806→        const spd = 0.65 + D * 0.7 + rng() * 0.25;
   807→        const baseX = THREE.MathUtils.clamp(x, -ROAD_HALF * 0.55 + 1, ROAD_HALF * 0.55 - 1) * 0.4;
   808→        const mphase = rng() * Math.PI * 2;
   809→        body = this.physics.addStaticBox({ x: 0.62, y: 0.75, z: 0.6 }, { x, y: 0.75, z: zEnd + lz }, { userData: { type, moving: true } });
   810→        hitC = { y: 0.8, r: 1.4 };
   811→        chunk.anims.push((t) => {
   812→          const mx = baseX + Math.sin(t * spd + mphase) * amp;
   813→          mesh.position.x = THREE.MathUtils.clamp(mx, -5.2, 5.2);
   814→          mesh.position.y = Math.abs(Math.sin(t * 2.2 + phase)) * 0.22;
   815→          mesh.rotation.z = Math.sin(t * 1.6 + phase) * 0.07;
   816→          body.position.x = mesh.position.x;
   817→          body.aabbNeedsUpdate = true; // keep raycasts truthful for movers
   818→        });
   819→      } else {

   962→  _addStarLine(chunk, rng, D, gapLocal) {
   963→    const n = 6 + ((rng() * 3) | 0);
   964→    const pattern = rng();
   965→    const startX = (rng() - 0.5) * 8;
   966→    const startZ = -(6 + rng() * (CHUNK_LEN - 16));
   967→    const spacing = 2.3;
   968→    for (let i = 0; i < n; i++) {
   969→      let x = startX, y = 1.35, z = startZ - i * spacing;
   970→      if (pattern < 0.30) {
   971→        // classic arc (floats over gaps)
   972→        y = 1.35 + Math.sin((i / (n - 1)) * Math.PI) * 2.2;
   973→        x = startX + Math.sin((i / (n - 1)) * Math.PI) * 1.6;
   974→      } else if (pattern < 0.55) {
   975→        // sine weave
   976→        x = startX + Math.sin(i * 0.85) * 2.4;
   977→      } else if (pattern < 0.72) {
   978→        // tight zigzag (harder, richer)
   979→        y = 1.35 + Math.sin((i / (n - 1)) * Math.PI) * 1.2;
   980→        x = startX + ((i % 2 === 0) ? -2.7 : 2.7);
   981→      } else if (pattern < 0.86) {
   982→        // double arc over a longer span
   983→        y = 1.35 + Math.abs(Math.sin((i / (n - 1)) * Math.PI * 2)) * 2.4;
   984→        x = startX + Math.sin((i / (n - 1)) * Math.PI * 2) * 2.0;
   985→      }
   986→      // ground-hugging patterns (weave / zigzag) skip gap sections;
   987→      // arcs and double arcs are high enough to float over gaps
   988→      const floatsOverGap = pattern < 0.30 || (pattern >= 0.72 && pattern < 0.86);
   989→      if (!floatsOverGap && this._inGap(gapLocal, z, 1)) continue;

   991→      const phase = rng() * 9;

  1166→  /** Sphere test for banana missiles: returns {body, x, y, z} of the first
  1167→   *  live obstacle whose hit-sphere contains the point. */
  1168→  missileHitTest(x, y, z) {
  1169→    for (const chunk of this.chunks) {
  1170→      const cz = chunk.group.position.z;
  1171→      for (const ob of chunk.obstacles) {
  1172→        if (ob.dead || !ob.body.userData) continue;
  1173→        const wz = ob.mesh.position.z + cz;
  1174→        const dz = z - wz;
  1175→        if (dz > 3 || dz < -3) continue; // cheap prefilter
  1176→        const c = ob.body.userData.hitC || { y: 1.0, r: 1.8 };
  1177→        const dx = x - ob.mesh.position.x;
  1178→        const dy = y - (ob.mesh.position.y + c.y);
  1179→        if (dx * dx + dy * dy + dz * dz < c.r * c.r) {
  1180→          return { body: ob.body, x: ob.mesh.position.x, y: ob.mesh.position.y + c.y, z: wz };
  1181→        }
  1182→      }
  1183→    }
  1184→    return null;

  1187→  /** Gentle auto-aim for the banana launcher: normalized direction toward
  1188→   *  the most threatening obstacle ahead (or null = straight forward). */
  1189→  aimAssist(px, py, pz) {
  1190→    let best = null, bestScore = 1e9;
  1191→    const my = py + 0.45;
  1192→    for (const chunk of this.chunks) {
  1193→      const cz = chunk.group.position.z;
  1194→      for (const ob of chunk.obstacles) {
  1195→        if (ob.dead) continue;
  1196→        const wz = ob.mesh.position.z + cz;
  1197→        const ahead = pz - wz;
  1198→        if (ahead < 4 || ahead > 68) continue;
  1199→        const dx = ob.mesh.position.x - px;
  1200→        if (Math.abs(dx) > 5.5) continue;
  1201→        const c = ob.body.userData.hitC || { y: 1.0, r: 1.8 };
  1202→        const dy = (ob.mesh.position.y + c.y) - my;
  1203→        const score = ahead + Math.abs(dx) * 1.5 + Math.abs(dy) * 2;
  1204→        if (score < bestScore) {
  1205→          bestScore = score;
  1206→          best = { x: dx, y: dy, z: -ahead };
  1207→        }
  1208→      }
  1209→    }
  1210→    if (!best) return null;
  1211→    const len = Math.hypot(best.x, best.y, best.z) || 1;
  1212→    return { x: best.x / len, y: best.y / len, z: best.z / len };
  1213→  }

  1235→    this.time += dt;
  1236→    const t = ctx.time;
  1237→    const pz = ctx.playerPos.z;
  1238→    const px = ctx.playerPos.x;
  1239→    const py = ctx.playerPos.y;
  1240→
  1241→    // ---- zone palette lerp ----
  1242→    this._lerpZone(dt);
  1243→
  1244→    // ---- recycle chunks that fell behind ----
  1245→    for (const chunk of this.chunks) {
  1246→      if (chunk.zEnd - CHUNK_LEN > pz + 26) this._recycle(chunk, ctx.difficulty);
  1247→    }
  1248→
  1249→    // ---- animate & collect ----
  1250→    for (const chunk of this.chunks) {
  1251→      for (const anim of chunk.anims) anim(t, dt);
  1252→      if (!ctx.canCollect) continue;
  1253→      for (const star of chunk.stars) {
  1254→        if (star.collected) continue;
  1255→        const dx = star.wx - px, dy = star.wy - py, dz = star.wz - pz;
  1256→        const d2 = dx * dx + dy * dy + dz * dz;
  1257→        if (ctx.superMode && d2 < 90) {
  1258→          const d = Math.sqrt(d2) || 1;
  1259→          const k = Math.min(1, (dt * 20) / d);
  1260→          star.wx += (px - star.wx) * k;
  1261→          star.wy += (py - star.wy) * k;
  1262→          star.wz += (pz - star.wz) * k;
  1263→          star.mesh.position.set(star.wx, star.wy, star.wz - chunk.zEnd);
  1264→          if (d < 2.4) this._collect(star, ctx);
  1265→        } else if (d2 < 2.9) {
  1266→          this._collect(star, ctx);
  1267→        }
  1268→      }
  1269→      // golden bananas (bigger radius + super magnet)
  1270→      for (const g of chunk.goldens) {
  1271→        if (g.collected) continue;
  1272→        const dx = g.wx - px, dy = g.wy - py, dz = g.wz - pz;
  1273→        const d2 = dx * dx + dy * dy + dz * dz;
  1274→        if (ctx.superMode && d2 < 42) {
  1275→          const d = Math.sqrt(d2) || 1;
  1276→          const k = Math.min(1, (dt * 16) / d);
  1277→          g.wx += (px - g.wx) * k;
  1278→          g.wy += (py - g.wy) * k;
  1279→          g.wz += (pz - g.wz) * k;
  1280→          g.mesh.position.set(g.wx, g.wy, g.wz - chunk.zEnd);
  1281→        }
  1282→        if (d2 < 3.4) {
  1283→          g.collected = true;
  1284→          g.mesh.visible = false;
  1285→          if (ctx.onGolden) ctx.onGolden(g);
  1286→        }
  1287→      }
  1288→    }
  1289→
  1290→    // ---- near-miss GRAZE detection (score for style) ----
  1291→    if (ctx.nearMissOK && ctx.onNearMiss) {
  1292→      for (const chunk of this.chunks) {
  1293→        for (const ob of chunk.obstacles) {
  1294→          if (ob.dead || ob.nearMissed) continue;
  1295→          const wz = ob.mesh.position.z + chunk.group.position.z;
  1296→          const rel = pz - wz;
  1297→          if (rel > 3.2) { ob.nearMissed = true; continue; } // passed cleanly
  1298→          if (rel > 0 && rel < 2.4) {
  1299→            const dx = Math.abs(px - ob.mesh.position.x);
  1300→            if (dx < (ob.body.userData.nmDist || 2.05)) {
  1301→              ob.nearMissed = true;
  1302→              ctx.onNearMiss(px, py + 0.6, pz);
  1303→            }
  1304→          }
  1305→        }
  1306→      }
  1307→    }
  1308→
  1309→    // ---- toast barrage ----
  1310→    this._updateToasts(dt, ctx, px, py, pz);
  1311→
  1312→    // ---- world-fixed grid floor follows the runner ----
  1313→    this.grid.position.z = pz;
  1314→    this.gridTex.offset.y = -pz / 10;
  1315→
  1316→    // ---- sky follows camera ----
  1317→    const cp = ctx.camera.position;
  1318→    this.sun.position.set(cp.x * 0.25, 30 + cp.y * 0.1, cp.z - 300);
  1319→    this.starfield.position.copy(cp);
  1320→  }
  1321→
  1322→  _collect(star, ctx) {
  1323→    star.collected = true;
  1324→    star.mesh.visible = false;
  1325→    if (ctx.onStar) ctx.onStar(star);
  1326→  }
