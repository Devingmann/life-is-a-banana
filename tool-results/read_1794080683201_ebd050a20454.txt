     5→/**
     6→ * Environment
     7→ * -----------
     8→ * Procedural chunk manager for the infinite "Highway of Life".
     9→ *  - Recycled 50-unit road chunks with neon edges, void gaps & warning strips
    10→ *  - Obstacles: Monkeys, Gorilla Bosses, Electric Toasters, the Giant Blender
    11→ *    of Fate — all merged single-draw-call meshes with cannon hitboxes
    12→ *  - Potassium stars (distance-collected + magnetized in Super Yellow Mode)
    13→ *  - Banana-peel slip traps (thin physics triggers → chaotic friction loss)
    14→ *  - Cyber-tropical scenery: neon palms, floating banana moons, neon arches,
    15→ *    synthwave grid floor, striped sun, starfield
    16→ */
    17→
    18→const CHUNK_LEN = 50;
    19→const CHUNK_COUNT = 10;
    20→const ROAD_W = 14;
    21→const START_Z = 20;
    22→
    23→const COL = {
    24→  banana: 0xffe135, magenta: 0xff2e92, pink: 0xff6ec7, teal: 0x00ffa3,
    25→  gold: 0xffd24a, purple: 0x9b5de5, trunk: 0x241338, frond: 0x0d6b4f,
    26→};
    27→
    28→/* ---------- tiny geometry helper: tint + transform + merge ---------- */
    29→const _m4 = new THREE.Matrix4();
    30→const _q = new THREE.Quaternion();
    31→const _e = new THREE.Euler();
    32→const _v = new THREE.Vector3();
    33→const _s = new THREE.Vector3();
    34→
    35→function part(geoSrc, hex, pos = [0, 0, 0], rot = [0, 0, 0], scl = [1, 1, 1]) {
    36→  const g = geoSrc.clone();
    37→  _e.set(rot[0], rot[1], rot[2]);
    38→  _q.setFromEuler(_e);
    39→  _v.set(pos[0], pos[1], pos[2]);
    40→  _s.set(scl[0], scl[1], scl[2]);
    41→  _m4.compose(_v, _q, _s);
    42→  g.applyMatrix4(_m4);
    43→  const n = g.attributes.position.count;
    44→  const c = new THREE.Color(hex);
    45→  const arr = new Float32Array(n * 3);
    46→  for (let i = 0; i < n; i++) { arr[i * 3] = c.r; arr[i * 3 + 1] = c.g; arr[i * 3 + 2] = c.b; }
    47→  g.setAttribute('color', new THREE.BufferAttribute(arr, 3));
    48→  return g;
    49→}

    60→    this.chunks = [];
    61→    this.chunkIndex = 0;

    67→
    68→    this._buildSharedAssets(bananaTemplate);
    69→    this._buildSky();
    70→
    71→    for (let i = 0; i < CHUNK_COUNT; i++) {
    72→      const chunk = { group: new THREE.Group(), zEnd: 0, allBodies: [], obstacles: [], stars: [], anims: [], gapSpan: null, disposables: [] };
    73→      this.group.add(chunk.group);
    74→      this.chunks.push(chunk);
    75→    }
    76→    this.reset();
    77→  }

   139→  _makeRoadTexture() {

   141→    c.width = 256; c.height = 512;
   142→    const ctx = c.getContext('2d');
   143→    ctx.fillStyle = '#171029';
   144→    ctx.fillRect(0, 0, 256, 512);
   145→    for (let i = 0; i < 1100; i++) {
   146→      ctx.fillStyle = Math.random() < 0.5 ? 'rgba(255,255,255,0.035)' : 'rgba(0,0,0,0.3)';
   147→      ctx.fillRect(Math.random() * 256, Math.random() * 512, 2, 2);
   148→    }
   149→    ctx.strokeStyle = 'rgba(255,46,146,0.22)';
   150→    ctx.lineWidth = 2;

   152→    for (let y = 0; y <= 512; y += 64) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(256, y); ctx.stroke(); }

   186→  _makeStarGeometry() {
   187→    const shape = new THREE.Shape();
   188→    const outer = 0.52, inner = 0.22;
   189→    for (let i = 0; i < 10; i++) {
   190→      const r = i % 2 === 0 ? outer : inner;
   191→      const a = (i / 10) * Math.PI * 2 - Math.PI / 2;
   192→      const x = Math.cos(a) * r, y = Math.sin(a) * r;
   193→      if (i === 0) shape.moveTo(x, y); else shape.lineTo(x, y);
   194→    }
   195→    shape.closePath();
   196→    const geo = new THREE.ExtrudeGeometry(shape, {
   197→      depth: 0.2, bevelEnabled: true, bevelThickness: 0.05, bevelSize: 0.05, bevelSegments: 1,
   198→    });
   199→    geo.center();
   200→    return geo;
   201→  }

   203→  _buildPalm() {
   204→    const trunk = new THREE.CylinderGeometry(0.16, 0.3, 5.4, 7);
   205→    const coco = new THREE.SphereGeometry(0.22, 8, 6);
   206→    const ring = new THREE.CylinderGeometry(0.33, 0.33, 0.09, 8);
   207→    const frond = new THREE.PlaneGeometry(3.1, 0.8, 4, 1);
   208→    // bend the frond into a droopy leaf
   209→    const fp = frond.attributes.position;
   210→    for (let i = 0; i < fp.count; i++) {
   211→      const x = fp.getX(i);
   212→      fp.setY(i, fp.getY(i) * (1 - Math.pow(Math.abs(x) / 1.55, 2) * 0.55));
   213→      fp.setZ(i, -Math.pow(Math.abs(x) / 1.55, 2) * 0.9);
   214→    }
   215→    frond.computeVertexNormals();
   216→
   217→    const silParts = [part(trunk, COL.trunk, [0, 2.7, 0])];
   218→    const ringParts = [];
   219→    [[1.2], [2.4], [3.6]].forEach(([h]) => ringParts.push(part(ring, 0, [0, h, 0])));
   220→    const cocoParts = [
   221→      part(coco, 0x3a2153, [0.3, 5.25, 0.12]),
   222→      part(coco, 0x3a2153, [-0.25, 5.3, -0.18]),
   223→    ];
   224→    const frondParts = [];
   225→    for (let i = 0; i < 7; i++) {
   226→      const a = (i / 7) * Math.PI * 2;
   227→      const dir = [Math.cos(a) * 1.35, 5.4, Math.sin(a) * 1.35];
   228→      frondParts.push(part(frond, 0, dir, [0, -a, -0.42], [1, 1, 1]));
   229→    }
   230→    this.geoPalmSil = mergeGeometries([...silParts, ...cocoParts], false);
   231→    this.geoPalmRings = mergeGeometries(ringParts, false);
   232→    this.geoPalmFronds = mergeGeometries(frondParts, false);
   233→    this.matPalmSil = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.9 });
   234→    this.matPalmRings = new THREE.MeshStandardMaterial({ color: COL.pink, emissive: COL.magenta, emissiveIntensity: 1.1 });
   235→    this.matPalmFronds = new THREE.MeshStandardMaterial({
   236→      color: 0x0e2b1f, emissive: COL.teal, emissiveIntensity: 0.65, roughness: 0.6, side: THREE.DoubleSide,
   237→    });
   238→  }

   329→  _buildSky() {
   330→    // striped synthwave sun
   331→    const c = document.createElement('canvas');
   332→    c.width = c.height = 256;
   333→    const ctx = c.getContext('2d');
   334→    const grad = ctx.createLinearGradient(0, 0, 0, 256);
   335→    grad.addColorStop(0, '#ffe95c');
   336→    grad.addColorStop(0.45, '#ff9d2e');
   337→    grad.addColorStop(1, '#ff2e92');
   338→    ctx.fillStyle = grad;
   339→    ctx.beginPath(); ctx.arc(128, 128, 120, 0, Math.PI * 2); ctx.fill();
   340→    ctx.globalCompositeOperation = 'destination-out';
   341→    let y = 128;
   342→    for (let i = 0; i < 6; i++) {
   343→      const h = 3 + i * 2.6;
   344→      ctx.fillRect(0, y, 256, h);
   345→      y += h + 20 - i * 2;
   346→    }
   347→    const sunTex = new THREE.CanvasTexture(c);
   348→    sunTex.colorSpace = THREE.SRGBColorSpace;
   349→    this.sun = new THREE.Mesh(
   350→      new THREE.PlaneGeometry(120, 120),
   351→      new THREE.MeshBasicMaterial({ map: sunTex, transparent: true, fog: false, depthWrite: false, blending: THREE.AdditiveBlending })
   352→    );
   353→    this.sun.renderOrder = -10;
   354→    this.scene.add(this.sun);
   355→
   356→    // starfield
   357→    const N = 700;
   358→    const pos = new Float32Array(N * 3);
   359→    const col = new Float32Array(N * 3);
   360→    const palette = [new THREE.Color(0xffffff), new THREE.Color(0xffd6f0), new THREE.Color(0xb6fff0), new THREE.Color(0xfff3b0)];
   361→    for (let i = 0; i < N; i++) {
   362→      const a = Math.random() * Math.PI * 2;
   363→      const el = Math.asin(Math.random() * 0.95); // upper hemisphere bias
   364→      const r = 280 + Math.random() * 50;
   365→      pos[i * 3] = Math.cos(a) * Math.cos(el) * r;
   366→      pos[i * 3 + 1] = Math.sin(el) * r * 0.9 + 10;
   367→      pos[i * 3 + 2] = Math.sin(a) * Math.cos(el) * r;
   368→      const c2 = palette[(Math.random() * palette.length) | 0];
   369→      col[i * 3] = c2.r; col[i * 3 + 1] = c2.g; col[i * 3 + 2] = c2.b;
   370→    }
   371→    const g = new THREE.BufferGeometry();
   372→    g.setAttribute('position', new THREE.BufferAttribute(pos, 3));
   373→    g.setAttribute('color', new THREE.BufferAttribute(col, 3));
   374→    this.starfield = new THREE.Points(g, new THREE.PointsMaterial({
   375→      size: 1.8, sizeAttenuation: false, vertexColors: true, transparent: true, opacity: 0.85, fog: false, depthWrite: false,
   376→    }));
   377→    this.starfield.renderOrder = -11;
   378→    this.scene.add(this.starfield);
   379→  }

   385→  reset() {
   386→    this.chunkIndex = 0;
   387→    for (let i = 0; i < this.chunks.length; i++) {
   388→      const chunk = this.chunks[i];
   389→      chunk.zEnd = START_Z - i * CHUNK_LEN;
   390→      this._generate(chunk, 0, i);
   391→    }
   392→    this.nextZ = START_Z - this.chunks.length * CHUNK_LEN;
   393→  }
   394→
   395→  _recycle(chunk, difficulty) {
   396→    chunk.zEnd = this.nextZ;
   397→    this.nextZ -= CHUNK_LEN;
   398→    this._generate(chunk, difficulty, this.chunkIndex++);
   399→  }

   401→  _generate(chunk, D, index) {
   402→    // ---- clear previous content ----
   403→    for (const b of chunk.allBodies) this.physics.removeBody(b);
   404→    chunk.allBodies.length = 0;
   405→    chunk.obstacles.length = 0;
   406→    chunk.stars.length = 0;
   407→    chunk.anims.length = 0;
   408→    for (const d of chunk.disposables) d.dispose();
   409→    chunk.disposables.length = 0;
   410→    chunk.group.clear();
   411→    chunk.gapSpan = null;
   412→
   413→    const zEnd = chunk.zEnd;
   414→    chunk.group.position.z = zEnd; // children use local z in [-CHUNK_LEN, 0]
   415→
   416→    const rng = Math.random;
   417→    const safeRunway = index < 2;
   418→
   419→    // ---- void gap layout ----
   420→    let gapLocal = null;
   421→    if (!safeRunway && index > 3 && rng() < Math.min(0.07 + D * 0.24, 0.33)) {
   422→      const gapLen = 7 + D * 5;
   423→      const gz = -(12 + rng() * (CHUNK_LEN - 26 - gapLen));
   424→      gapLocal = { near: gz, far: gz - gapLen };
   425→    }
   426→
   427→    // ---- solid road segments ----
   428→    const segments = [];
   429→    if (gapLocal) {
   430→      segments.push({ z0: 0, z1: gapLocal.near });
   431→      segments.push({ z0: gapLocal.far, z1: -CHUNK_LEN });
   432→    } else {
   433→      segments.push({ z0: 0, z1: -CHUNK_LEN });
   434→    }
   435→    for (const seg of segments) {
   436→      if (seg.z0 - seg.z1 < 1) continue; // skip degenerate segments (length is z0 - z1)
   437→      const len = seg.z0 - seg.z1;
   438→      const mid = (seg.z0 + seg.z1) / 2;
   439→      const worldMid = zEnd + mid;
   440→      // road plane
   441→      const geo = new THREE.PlaneGeometry(ROAD_W, len);
   442→      const uv = geo.attributes.uv;
   443→      for (let i = 0; i < uv.count; i++) uv.setY(i, uv.getY(i) * (len / 10));
   444→      const road = new THREE.Mesh(geo, this.matRoad);
   445→      road.rotation.x = -Math.PI / 2;
   446→      road.position.set(0, 0, mid);
   447→      road.receiveShadow = true;
   448→      chunk.group.add(road);
   449→      chunk.disposables.push(geo);
   450→      // physics slab (top at y=0)
   451→      chunk.allBodies.push(this.physics.addStaticBox(
   452→        { x: ROAD_W / 2, y: 0.5, z: len / 2 }, { x: 0, y: -0.5, z: worldMid }, { kind: 'ground' }
   453→      ));
   454→      // neon edge strips
   455→      for (const sx of [-ROAD_W / 2 + 0.15, ROAD_W / 2 - 0.15]) {
   456→        const strip = new THREE.Mesh(this.geoEdge, this.matEdge);
   457→        strip.scale.set(0.3, 0.17, len);
   458→        strip.position.set(sx, 0.08, mid);
   459→        chunk.group.add(strip);
   460→      }
   461→    }
   462→
   463→    // gap warning strips + span bookkeeping
   464→    if (gapLocal) {
   465→      for (const gz of [gapLocal.near + 0.4, gapLocal.far - 0.4]) {
   466→        const w = new THREE.Mesh(this.geoWarn, this.matWarn);
   467→        w.position.set(0, 0.045, gz);
   468→        chunk.group.add(w);
   469→      }
   470→      chunk.gapSpan = { zNear: zEnd + gapLocal.near, zFar: zEnd + gapLocal.far };
   471→    }
   472→
   473→    if (safeRunway) {
   474→      this._addScenery(chunk, rng);
   475→      this._addStarLine(chunk, rng, D, null);
   476→      return;
   477→    }
   478→
   479→    // ---- slip zones (banana peel traps) ----
   480→    const slipZones = [];
   481→    if (rng() < 0.22 + D * 0.16) {
   482→      const patchZ = -(8 + rng() * (CHUNK_LEN - 18));
   483→      if (!this._inGap(gapLocal, patchZ, 6)) {
   484→        const px = (rng() - 0.5) * 6.4;
   485→        const half = { x: 2.7, y: 0.09, z: 4.2 };
   486→        const body = this.physics.addStaticBox(half, { x: px, y: 0.1, z: zEnd + patchZ }, {
   487→          kind: 'slipzone', material: this.physics.groundMaterial, userData: { chunk, id: rng() },
   488→        });
   489→        chunk.allBodies.push(body);
   490→        slipZones.push(patchZ);
   491→        // scattered peel bits
   492→        const nPeels = 7 + ((rng() * 3) | 0);
   493→        for (let i = 0; i < nPeels; i++) {
   494→          const peel = new THREE.Mesh(this.geoPeel, this.matPeel);
   495→          peel.position.set(px + (rng() - 0.5) * 4.6, 0.06, patchZ + (rng() - 0.5) * 7);
   496→          peel.rotation.set(0, rng() * Math.PI * 2, (rng() - 0.5) * 0.25);
   497→          peel.scale.setScalar(0.8 + rng() * 0.5);
   498→          chunk.group.add(peel);
   499→        }
   500→      }
   501→    }
   502→
   503→    // ---- obstacles ----
   504→    const count = 2 + Math.round(D * 3) + (D > 0.55 && rng() < 0.35 ? 1 : 0);
   505→    const placed = [];
   506→    for (let i = 0; i < count; i++) {
   507→      let lz = null;
   508→      for (let attempt = 0; attempt < 26; attempt++) {
   509→        const z = -(5 + rng() * (CHUNK_LEN - 10));
   510→        if (this._inGap(gapLocal, z, 7)) continue;
   511→        if (slipZones.some((s) => Math.abs(s - z) < 6)) continue;
   512→        if (placed.some((s) => Math.abs(s - z) < 8.5)) continue;
   513→        lz = z;
   514→        break;
   515→      }
   516→      if (lz === null) continue;
   517→      placed.push(lz);
   518→      const type = this._pickObstacleType(D, rng, placed.length === 1);
   519→      this._addObstacle(chunk, type, lz, rng, D);
   520→    }
   521→
   522→    // ---- potassium star lines ----
   523→    if (rng() < 0.68) this._addStarLine(chunk, rng, D, gapLocal);
   524→
   525→    // ---- scenery ----
   526→    this._addScenery(chunk, rng);
   527→  }
   528→
   529→  _inGap(gapLocal, z, margin) {
   530→    return !!gapLocal && z < gapLocal.near + margin && z > gapLocal.far - margin;
   531→  }

   533→  _pickObstacleType(D, rng, first) {
   534→    const entries = [
   535→      ['monkey', 10],
   536→      ['toaster', D > 0.08 ? 7 : 0],
   537→      ['blender', D > 0.18 ? 4 + D * 5 : 0],
   538→      ['gorilla', D > 0.3 && first ? 3.5 : 0],
   539→    ];
   540→    let total = 0;
   541→    for (const [, w] of entries) total += w;
   542→    let r = rng() * total;
   543→    for (const [t, w] of entries) { r -= w; if (r <= 0) return t; }
   544→    return 'monkey';
   545→  }

   547→  _addObstacle(chunk, type, lz, rng, D) {
   548→    const zEnd = chunk.zEnd;
   549→    let mesh, body, damage = 1, knock = 7;
   550→
   551→    if (type === 'monkey') {
   552→      mesh = new THREE.Mesh(this.geoMonkey, this.matFlat);
   553→      mesh.castShadow = true;
   554→      const x = (rng() - 0.5) * 8.8;
   555→      mesh.position.set(x, 0, lz);
   556→      const phase = rng() * 9;
   557→      chunk.anims.push((t) => {
   558→        mesh.position.y = Math.abs(Math.sin(t * 2.2 + phase)) * 0.22;
   559→        mesh.rotation.z = Math.sin(t * 1.6 + phase) * 0.07;
   560→      });
   561→      body = this.physics.addStaticBox({ x: 0.62, y: 0.75, z: 0.6 }, { x, y: 0.75, z: zEnd + lz }, { userData: { type } });
   562→    } else if (type === 'gorilla') {
   563→      mesh = new THREE.Group();
   564→      const g = new THREE.Mesh(this.geoGorilla, this.matFlat);
   565→      g.castShadow = true;
   566→      mesh.add(g);
   567→      const arms = [];
   568→      for (const sx of [0.95, -0.95]) {
   569→        const arm = new THREE.Group();
   570→        const am = new THREE.Mesh(this.geoGorillaArm, this.matGorillaArm);
   571→        am.position.y = -0.65;
   572→        am.castShadow = true;
   573→        const fist = new THREE.Mesh(this.geoGorillaFist, this.matGorillaArm);
   574→        fist.position.y = -1.35;
   575→        arm.add(am, fist);
   576→        arm.position.set(sx, 1.95, 0.15);
   577→        mesh.add(arm);
   578→        arms.push(arm);
   579→      }
   580→      const x = rng() < 0.5 ? -3.1 : 3.1;
   581→      mesh.position.set(x, 0, lz);
   582→      const phase = rng() * 9;
   583→      chunk.anims.push((t) => {
   584→        const thump = Math.max(0, Math.sin(t * 0.9 + phase));
   585→        const swing = Math.sin(t * 3 + phase) * 0.45;
   586→        arms[0].rotation.x = thump > 0.94 ? -2.4 : swing;
   587→        arms[1].rotation.x = thump > 0.94 ? -2.4 : -swing;
   588→        mesh.position.y = Math.abs(Math.sin(t * 2 + phase)) * 0.12;
   589→      });
   590→      damage = 2; knock = 11;
   591→      body = this.physics.addStaticBox({ x: 1.05, y: 1.35, z: 0.85 }, { x, y: 1.35, z: zEnd + lz }, { userData: { type } });
   592→    } else if (type === 'toaster') {
   593→      mesh = new THREE.Group();
   594→      const tm = new THREE.Mesh(this.geoToaster, this.matToaster);
   595→      tm.castShadow = true;
   596→      const slots = new THREE.Mesh(this.geoSlots, this.matSlots);
   597→      slots.position.y = 0.0;
   598→      const toasts = [];
   599→      for (const tx of [0.3, -0.3]) {
   600→        const toast = new THREE.Mesh(this.geoToast, this.matToast);
   601→        toast.position.set(tx, 0.85, 0);
   602→        toasts.push(toast);
   603→        mesh.add(toast);
   604→      }
   605→      mesh.add(tm, slots);
   606→      const x = (rng() - 0.5) * 8.6;
   607→      mesh.position.set(x, 0, lz);
   608→      const phase = rng() * 9;
   609→      chunk.anims.push((t) => {
   610→        const pop = Math.max(0, Math.sin(t * 1.7 + phase));
   611→        const h = 0.85 + Math.pow(pop, 3) * 0.75;
   612→        toasts[0].position.y = h;
   613→        toasts[1].position.y = 0.85 + Math.pow(Math.max(0, Math.sin(t * 1.7 + phase + 0.7)), 3) * 0.75;
   614→        slots.material.emissiveIntensity = 1.05 + Math.pow(pop, 4) * 1.6;
   615→      });
   616→      body = this.physics.addStaticBox({ x: 0.75, y: 0.55, z: 0.62 }, { x, y: 0.55, z: zEnd + lz }, { userData: { type } });
   617→    } else {
   618→      // blender of fate
   619→      mesh = new THREE.Group();
   620→      const glass = new THREE.Mesh(this.geoBlenderGlass, this.matBlenderGlass);
   621→      glass.position.y = 2.35;
   622→      const metal = new THREE.Mesh(this.geoBlenderMetal, this.matBlenderMetal);
   623→      metal.castShadow = true;
   624→      const blades = new THREE.Mesh(this.geoBlades, this.matBlades);
   625→      blades.position.y = 1.0;
   626→      const smoothie = new THREE.Mesh(this.geoSmoothie, this.matSmoothie);
   627→      smoothie.position.y = 1.5;
   628→      mesh.add(glass, metal, blades, smoothie);
   629→      const x = (rng() - 0.5) * 4.6;
   630→      mesh.position.set(x, 0, lz);
   631→      chunk.anims.push((t, dt) => {
   632→        blades.rotation.y += 15 * dt;
   633→        smoothie.rotation.y -= 4 * dt;
   634→      });
   635→      knock = 13;
   636→      body = this.physics.addStaticCylinder(2.0, 4.1, { x, y: 2.05, z: zEnd + lz }, { userData: { type } });
   637→    }
   638→
   639→    chunk.group.add(mesh);
   640→    body.userData.kind = 'obstacle';
   641→    body.userData.damage = damage;
   642→    body.userData.knock = knock;
   643→    body.userData.mesh = mesh;
   644→    body.userData.chunk = chunk;
   645→    body.userData.dead = false;
   646→    const obstacle = { mesh, body, dead: false };
   647→    chunk.obstacles.push(obstacle);
   648→    chunk.allBodies.push(body);
   649→  }

   651→  _addStarLine(chunk, rng, D, gapLocal) {
   652→    const n = 6 + ((rng() * 3) | 0);
   653→    const pattern = rng();
   654→    const startX = (rng() - 0.5) * 8;
   655→    const startZ = -(6 + rng() * (CHUNK_LEN - 16));
   656→    const spacing = 2.3;
   657→    for (let i = 0; i < n; i++) {
   658→      let x = startX, y = 1.35, z = startZ - i * spacing;
   659→      if (pattern < 0.36) {
   660→        y = 1.35 + Math.sin((i / (n - 1)) * Math.PI) * 2.2; // arc
   661→        x = startX + Math.sin((i / (n - 1)) * Math.PI) * 1.6;
   662→      } else if (pattern < 0.62) {
   663→        x = startX + Math.sin(i * 0.85) * 2.4; // sine weave
   664→      }
   665→      // straight lines skip gap sections; arcs are allowed to float over gaps
   666→      if (pattern >= 0.36 && this._inGap(gapLocal, z, 1)) continue;
   667→      const mesh = new THREE.Mesh(this.geoStar, this.matStar);
   668→      const phase = rng() * 9;
   669→      const baseY = y;
   670→      mesh.position.set(x, y, z);
   671→      mesh.rotation.z = rng() * Math.PI;
   672→      chunk.group.add(mesh);
   673→      const star = { mesh, wx: x, wy: y, wz: chunk.zEnd + z, baseY, phase, collected: false };
   674→      chunk.stars.push(star);
   675→      chunk.anims.push((t, dt) => {
   676→        if (star.collected) return;
   677→        mesh.rotation.y += 2.6 * dt;
   678→        mesh.position.y = star.baseY + Math.sin(t * 2.4 + phase) * 0.26;
   679→        star.wy = mesh.position.y;
   680→      });
   681→    }
   682→  }

   684→  _addScenery(chunk, rng) {
   685→    // neon palms
   686→    const palmsPerSide = 2 + ((rng() * 4) | 0);
   687→    for (let side = 0; side < 2; side++) {
   688→      for (let i = 0; i < palmsPerSide; i++) {
   689→        const sgn = side === 0 ? -1 : 1;
   690→        const x = sgn * (10.5 + rng() * 13);
   691→        const z = -(4 + rng() * (CHUNK_LEN - 8));
   692→        const palm = new THREE.Group();
   693→        const sil = new THREE.Mesh(this.geoPalmSil, this.matPalmSil);
   694→        sil.castShadow = true;
   695→        const rings = new THREE.Mesh(this.geoPalmRings, this.matPalmRings);
   696→        const fronds = new THREE.Mesh(this.geoPalmFronds, this.matPalmFronds);
   697→        palm.add(sil, rings, fronds);
   698→        palm.position.set(x, 0, z);
   699→        palm.rotation.y = rng() * Math.PI * 2;
   700→        palm.scale.setScalar(0.8 + rng() * 0.55);
   701→        chunk.group.add(palm);
   702→      }
   703→    }
   704→    // floating banana moons
   705→    if (rng() < 0.3 && this.moonTemplate) {
   706→      const moon = this.moonTemplate.clone(true);
   707→      moon.traverse((o) => { if (o.isMesh) o.material = this.moonMat; });
   708→      const sgn = rng() < 0.5 ? -1 : 1;
   709→      moon.position.set(sgn * (16 + rng() * 26), 17 + rng() * 14, -(10 + rng() * (CHUNK_LEN - 20)));
   710→      moon.scale.setScalar(5.5 + rng() * 4);
   711→      moon.rotation.z = (rng() - 0.5) * 1.2;
   712→      chunk.group.add(moon);
   713→      const rot = 0.15 + rng() * 0.25;
   714→      chunk.anims.push((t, dt) => { moon.rotation.y += rot * dt; });
   715→    }
   716→    // neon arch
   717→    if (rng() < 0.24) {
   718→      const arch = new THREE.Mesh(this.geoArch, rng() < 0.5 ? this.matArchPink : this.matArchTeal);
   719→      arch.position.set(0, 0, -(8 + rng() * (CHUNK_LEN - 16)));
   720→      chunk.group.add(arch);
   721→    }
   722→  }

   728→  /** Remove an obstacle (Super Yellow Mode smash). */
   729→  smashObstacle(body) {
   730→    if (!body.userData || body.userData.dead) return null;
   731→    body.userData.dead = true;
   732→    const mesh = body.userData.mesh;
   733→    const pos = mesh ? mesh.getWorldPosition(new THREE.Vector3()) : new THREE.Vector3();
   734→    if (mesh) mesh.visible = false;
   735→    this.physics.removeBody(body);
   736→    const ob = (body.userData.chunk && body.userData.chunk.obstacles || []).find((o) => o.body === body);
   737→    if (ob) ob.dead = true;
   738→    return pos;
   739→  }

   741→  /** A z position ahead of fromZ guaranteed to be solid road. */
   742→  findSafeZ(fromZ) {
   743→    let z = fromZ - 13;
   744→    for (let iter = 0; iter < 4; iter++) {
   745→      for (const chunk of this.chunks) {
   746→        if (!chunk.gapSpan) continue;
   747→        const g = chunk.gapSpan;
   748→        if (z < g.zNear + 5 && z > g.zFar - 5) z = g.zFar - 7;
   749→      }
   750→    }
   751→    return z;
   752→  }

   754→  /**
   755→   * @param {number} dt
   756→   * @param {{playerPos:THREE.Vector3, camera:THREE.Camera, canCollect:boolean,
   757→   *          superMode:boolean, onStar:Function, difficulty:number, time:number}} ctx
   758→   */
   759→  update(dt, ctx) {
   760→    this.time += dt;
   761→    const t = ctx.time;
   762→    const pz = ctx.playerPos.z;
   763→    const px = ctx.playerPos.x;
   764→    const py = ctx.playerPos.y;
   765→
   766→    // ---- recycle chunks that fell behind ----
   767→    for (const chunk of this.chunks) {
   768→      if (chunk.zEnd - CHUNK_LEN > pz + 26) this._recycle(chunk, ctx.difficulty);
   769→    }
   770→
   771→    // ---- animate & collect ----
   772→    for (const chunk of this.chunks) {
   773→      for (const anim of chunk.anims) anim(t, dt);
   774→      if (!ctx.canCollect) continue;
   775→      for (const star of chunk.stars) {
   776→        if (star.collected) continue;
   777→        const dx = star.wx - px, dy = star.wy - py, dz = star.wz - pz;
   778→        const d2 = dx * dx + dy * dy + dz * dz;
   779→        if (ctx.superMode && d2 < 90) {
   780→          const d = Math.sqrt(d2) || 1;
   781→          const k = Math.min(1, (dt * 20) / d);
   782→          star.wx += (px - star.wx) * k;
   783→          star.wy += (py - star.wy) * k;
   784→          star.wz += (pz - star.wz) * k;
   785→          star.mesh.position.set(star.wx, star.wy, star.wz - chunk.zEnd);
   786→          if (d < 2.4) this._collect(star, ctx);
   787→        } else if (d2 < 2.9) {
   788→          this._collect(star, ctx);
   789→        }
   790→      }
   791→    }
   792→
   793→    // ---- world-fixed grid floor follows the runner ----
   794→    this.grid.position.z = pz;
   795→    this.gridTex.offset.y = -pz / 10;
   796→
   797→    // ---- sky follows camera ----
   798→    const cp = ctx.camera.position;
   799→    this.sun.position.set(cp.x * 0.25, 30 + cp.y * 0.1, cp.z - 300);
   800→    this.starfield.position.copy(cp);
   801→  }
   802→
   803→  _collect(star, ctx) {
   804→    star.collected = true;
   805→    star.mesh.visible = false;
   806→    if (ctx.onStar) ctx.onStar(star);
   807→  }
